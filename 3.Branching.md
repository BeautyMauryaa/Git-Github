# Branches
git branch
git branch <name>
git branch -d <name>
git branch -D <name>
git branch -a
git branch -vv

# Switching
git switch <branch>
git switch -c <new-branch>

# Older syntax
git checkout <branch>
git checkout -b <new-branch>

# Merge
git merge <branch>
git merge --abort

# Rebase
git rebase <branch>
git rebase --abort
git rebase --continue
git rebase -i HEAD~N

# Remote branch
git push -u origin <branch>

# History
git log --oneline --graph --decorate --all
