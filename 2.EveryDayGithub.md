git status

# move the current version of a file into the staging area

```text
git add <file>
git add .
```

# staging selectively:

```text
login.js       changed
payment.js     changed
navbar.css     changed

git add navbar.css

Staged:
    navbar.css

Unstaged:
    login.js
    payment.js

git commit -m "Fix navbar styling"
```

# git diff:

- show unstaged changes in tracked files

git diff
git diff --staged

# restore:

git restore <file>
git restore --staged <file>

# records the staged state in git history

git commit -m "message"

# see commit history:

git log
git log --oneline
git log --graph --oneline --decorate --all

# want to inspect particular commit:

git show <commit>
ex: git show a71f4c2

git commit --amend

git clean -n
git clean -f

# remove tracked file

git rm <file>

- remove a tracked file and staged the deletion
- git show a71f4c2 and then commit : git commit -m "Remove obsolete file"

# rename/move tracked file

git mv <old> <new>

- rename/move tracked file
  git commit -m "Remove obsolete file"
  then git commit -m "Rename utility module"


# modify most recent commit:
git commit --amend 
- do carefully after pushing
